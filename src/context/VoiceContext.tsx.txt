import React, { createContext, useContext, useState, useEffect, useRef } from 'react';

interface VoiceContextType {
  isListening: boolean;
  isVoiceActive: boolean;
  codeWord: string;
  repetitions: number;
  toggleVoice: () => void;
  setCodeWord: (word: string) => void;
  setRepetitions: (num: number) => void;
  testVoice: () => void;
}

const VoiceContext = createContext<VoiceContextType | undefined>(undefined);

export const useVoice = () => {
  const context = useContext(VoiceContext);
  if (!context) {
    throw new Error('useVoice must be used within VoiceProvider');
  }
  return context;
};

export const VoiceProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [isVoiceActive, setIsVoiceActive] = useState(false);
  const [isListening, setIsListening] = useState(false);
  const [codeWord, setCodeWord] = useState('');
  const [repetitions, setRepetitions] = useState(3);
  const [recentPhrases, setRecentPhrases] = useState<string[]>([]);
  const recognitionRef = useRef<any>(null);

  useEffect(() => {
    const saved = localStorage.getItem('voiceSettings');
    if (saved) {
      const settings = JSON.parse(saved);
      setIsVoiceActive(settings.isVoiceActive || false);
      setCodeWord(settings.codeWord || '');
      setRepetitions(settings.repetitions || 3);
    }
  }, []);

  useEffect(() => {
    localStorage.setItem('voiceSettings', JSON.stringify({
      isVoiceActive,
      codeWord,
      repetitions
    }));
  }, [isVoiceActive, codeWord, repetitions]);

  const startListening = () => {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
      alert('Voice recognition not supported in this browser');
      return;
    }

    const SpeechRecognition = (window as any).webkitSpeechRecognition || (window as any).SpeechRecognition;
    const recognition = new SpeechRecognition();
    recognition.continuous = true;
    recognition.interimResults = true;
    recognition.lang = 'en-US';

    recognition.onresult = (event: any) => {
      const transcript = event.results[event.results.length - 1][0].transcript.toLowerCase();
      setRecentPhrases(prev => [...prev.slice(-repetitions + 1), transcript]);

      const allPhrases = [...recentPhrases.slice(-repetitions + 1), transcript];
      if (codeWord && allPhrases.length === repetitions) {
        const allMatch = allPhrases.every(phrase => phrase.includes(codeWord.toLowerCase()));
        if (allMatch) {
          window.dispatchEvent(new CustomEvent('voice-emergency-triggered'));
          setRecentPhrases([]);
        }
      }
    };

    recognition.start();
    recognitionRef.current = recognition;
    setIsListening(true);
  };

  const stopListening = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop();
      setIsListening(false);
    }
  };

  const toggleVoice = () => {
    const newState = !isVoiceActive;
    setIsVoiceActive(newState);
    if (newState) {
      startListening();
    } else {
      stopListening();
    }
  };

  const testVoice = () => {
    if (!codeWord) {
      alert('Please set a code word first');
      return;
    }
    alert(`Voice detection test. Say "${codeWord}" ${repetitions} times in a row to trigger emergency.`);
  };

  useEffect(() => {
    return () => {
      if (recognitionRef.current) {
        recognitionRef.current.stop();
      }
    };
  }, []);

  useEffect(() => {
    if (isVoiceActive) {
      startListening();
    } else {
      stopListening();
    }
  }, [isVoiceActive]);

  return (
    <VoiceContext.Provider value={{
      isListening,
      isVoiceActive,
      codeWord,
      repetitions,
      toggleVoice,
      setCodeWord,
      setRepetitions,
      testVoice
    }}>
      {children}
    </VoiceContext.Provider>
  );
};
